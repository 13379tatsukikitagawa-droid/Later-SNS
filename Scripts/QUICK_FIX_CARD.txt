  📁 Multiple files with toolbar errors
  
     Error: "Ambiguous use of 'toolbar(content:)'"
     Fix:   Add placement parameter
     
     Before:
     .toolbar {
         Button("Done") { }
     }
     
     After:
     .toolbar {
         ToolbarItem(placement: .navigationBarTrailing) {
             Button("Done") { }
         }
     }
  
  ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
  
  📁 HomeView.swift (lines 18, 19)
  
     Error: "Ambiguous use of 'shared'"
     Fix:   Fully qualify the type
     
     Before:
     let cache = shared
     
     After:
     let cache = ImageCacheManager.shared
